cmake_minimum_required(VERSION 3.16)
project(InjectorDiagnostic VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SOURCES src/ecu/ECU.cpp src/obd/OBD.cpp src/dtc/DTC.cpp src/live_data/LiveData.cpp src/injector/InjectorTester.cpp src/diagnosis/Diagnosis.cpp src/communication/SimulationCommunication.cpp src/settings/Settings.cpp)
add_executable(injector_diagnostic src/main.cpp ${SOURCES})
target_include_directories(injector_diagnostic PRIVATE src)
if(MSVC)
 target_compile_options(injector_diagnostic PRIVATE /W4 /permissive-)
else()
 target_compile_options(injector_diagnostic PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(injector_diagnostic_tests tests/test_main.cpp ${SOURCES})
target_include_directories(injector_diagnostic_tests PRIVATE src)
if(MSVC)
 target_compile_options(injector_diagnostic_tests PRIVATE /W4 /permissive-)
else()
 target_compile_options(injector_diagnostic_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME injector_diagnostic_tests COMMAND injector_diagnostic_tests)
